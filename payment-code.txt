<script>

let selectedPerson = "";
let selectedPersonImage = "";
let selectedAmount = 0;


/* OPEN PAYMENT */

function openPayment(name, image) {

    selectedPerson = name;
    selectedPersonImage = image;

    document.getElementById("selectedName").textContent =
        name;

    document.getElementById("selectedImage").src =
        image;

    document.getElementById("amountInput").value = "";

    document.getElementById("noteInput").value = "";

    document.getElementById("paymentModal").style.display =
        "flex";

}


/* CLOSE PAYMENT */

function closePayment() {

    document.getElementById("paymentModal").style.display =
        "none";

}


/* CONTINUE */

function continuePayment() {

    const amount =
        document.getElementById("amountInput").value;

    if (!amount || Number(amount) <= 0) {

        alert("Please enter a valid amount.");

        return;

    }

    selectedAmount = Number(amount);

    closePayment();

    document.getElementById("pinInput").value = "";

    document.getElementById("pinModal").style.display =
        "flex";

}


/* CLOSE PIN */

function closePin() {

    document.getElementById("pinModal").style.display =
        "none";

}


/* PROCESS PAYMENT */

function processPayment() {

    const pin =
        document.getElementById("pinInput").value;

    if (pin.length !== 4) {

        alert("Please enter a 4-digit demo PIN.");

        return;

    }

    /*
       DEMO ONLY

       No real bank transaction happens here.
    */

    closePin();

    document.getElementById("successMessage").textContent =
        "₹" + selectedAmount.toLocaleString() +
        " sent successfully to " +
        selectedPerson + ".";

    addTransaction();

    document.getElementById("successModal").style.display =
        "flex";

}


/* ADD TRANSACTION */

function addTransaction() {

    const transactions =
        document.getElementById("transactions");

    const transaction =
        document.createElement("div");

    transaction.className = "transaction";

    transaction.innerHTML = `

        <div class="transaction-icon">
            ↑
        </div>

        <div class="transaction-info">

            <h4>${escapeHTML(selectedPerson)}</h4>

            <p>Just now</p>

        </div>

        <div class="amount sent">
            - ₹${selectedAmount.toLocaleString()}
        </div>
    `;

    transactions.prepend(transaction);

}


/* CLOSE SUCCESS */

function closeSuccess() {

    document.getElementById("successModal").style.display =
        "none";

}


/* SEARCH CONTACTS */

function searchContacts() {

    const search =
        document.getElementById("searchInput")
        .value
        .toLowerCase();

    const contacts =
        document.querySelectorAll(".contact");

    contacts.forEach(function(contact) {

        const name =
            contact.dataset.name.toLowerCase();

        if (name.includes(search)) {

            contact.style.display = "flex";

        } else {

            contact.style.display = "none";

        }

    });

}


/* ESCAPE HTML */

function escapeHTML(text) {

    const div = document.createElement("div");

    div.textContent = text;

    return div.innerHTML;

}

</script>
