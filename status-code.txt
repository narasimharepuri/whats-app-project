<script>

/* STATUS DATA */

const statuses = [

    {
        name: "Rahul",
        image: "https://i.pravatar.cc/100?img=11",
        type: "text",
        content: "Enjoying the evening! 🌅",

    },

];

let currentStatus = 0;

let myStatus = null;

let progressTimer;


/* OPEN ADD STATUS */

function openAddStatus() {

    document.getElementById("addModal").style.display = "flex";

}


/* CLOSE ADD STATUS */

function closeAddStatus() {

    document.getElementById("addModal").style.display = "none";

}


/* POST STATUS */

function postStatus() {

    const text =
        document.getElementById("statusText").value.trim();

    if (text === "") {

        alert("Please enter a status.");

        return;

    }

    myStatus = {

        name: "My status",

        image: "https://i.pravatar.cc/100?img=12",

        type: "text",

        content: text,

        time: "Just now"

    };

    document.getElementById("myStatusText").textContent =
        "Just now";

    closeAddStatus();

    document.getElementById("statusText").value = "";

    alert("Status posted successfully!");

}


/* OPEN STATUS */

function openStatus(index) {

    currentStatus = index;

    showStatus();

    document.getElementById("viewer").style.display = "flex";

}


/* SHOW STATUS */

function showStatus() {

    clearInterval(progressTimer);

    let status;

    if (currentStatus === -1 && myStatus) {

        status = myStatus;

    } else {

        status = statuses[currentStatus];

    }

    document.getElementById("viewerName").textContent =
        status.name;

    document.getElementById("viewerTime").textContent =
        status.time;

    document.getElementById("viewerImage").src =
        status.image;


    const content =
        document.getElementById("viewerStatus");

    content.innerHTML = "";

    if (status.type === "image") {

        const image = document.createElement("img");

        image.src = status.content;

        content.appendChild(image);

    } else {

        const text = document.createElement("div");

        text.className = "text-status";

        text.textContent = status.content;

        content.appendChild(text);

    }


    /* DELETE BUTTON */

    const deleteButton =
        document.getElementById("deleteButton");

    if (currentStatus === -1 && myStatus) {

        deleteButton.style.display = "block";

    } else {

        deleteButton.style.display = "none";

    }


    /* PROGRESS BARS */

    const progressContainer =
        document.getElementById("progressContainer");

    progressContainer.innerHTML = "";

    let totalStatuses = statuses.length;

    if (myStatus) {

        totalStatuses++;

    }

    for (let i = 0; i < totalStatuses; i++) {

        const progress =
            document.createElement("div");

        progress.className = "progress";

        const fill =
            document.createElement("div");

        fill.className = "progress-fill";

        if (i < currentStatus) {

            fill.style.width = "100%";

        }

        progress.appendChild(fill);

        progressContainer.appendChild(progress);

    }


    /* TIMER */

    let width = 0;

    const currentProgress =
        progressContainer.children[
            currentStatus === -1 ? totalStatuses - 1 : currentStatus
        ];

    if (currentProgress) {

        const fill =
            currentProgress.querySelector(".progress-fill");

        progressTimer = setInterval(function() {

            width += 1;

            fill.style.width = width + "%";

            if (width >= 100) {

                clearInterval(progressTimer);

                nextStatus();

            }

        }, 50);

    }

}


/* NEXT */

function nextStatus() {

    clearInterval(progressTimer);

    if (myStatus && currentStatus < 0) {

        currentStatus = 0;

    } else {

        currentStatus++;

    }

    if (currentStatus >= statuses.length) {

        if (myStatus) {

            currentStatus = -1;

            showStatus();

        } else {

            closeViewer();

        }

    } else {

        showStatus();

    }

}


/* PREVIOUS */

function previousStatus() {

    clearInterval(progressTimer);

    if (currentStatus === -1) {

        currentStatus = statuses.length - 1;

    } else {

        currentStatus--;

        if (currentStatus < 0) {

            if (myStatus) {

                currentStatus = -1;

            } else {

                currentStatus = statuses.length - 1;

            }

        }

    }

    showStatus();

}


/* DELETE MY STATUS */

function deleteMyStatus() {

    myStatus = null;

    document.getElementById("myStatusText").textContent =
        "Click to add status";

    closeViewer();

    alert("Your status has been deleted.");

}


/* CLOSE VIEWER */

function closeViewer() {

    clearInterval(progressTimer);

    document.getElementById("viewer").style.display = "none";

}


/* KEYBOARD CONTROLS */

document.addEventListener("keydown", function(event) {

    const viewer =
        document.getElementById("viewer");

    if (viewer.style.display === "flex") {

        if (event.key === "ArrowRight") {

            nextStatus();

        }

        if (event.key === "ArrowLeft") {

            previousStatus();

        }

        if (event.key === "Escape") {

            closeViewer();

        }

    }

});

</script>
