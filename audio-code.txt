<script>

let mediaRecorder;

let audioChunks = [];

let audioBlob;

let audioURL;

let recordingSeconds = 0;

let recordingTimer;


/* START RECORDING */

async function startRecording() {

    try {

        const stream =
            await navigator.mediaDevices.getUserMedia({
                audio: true
            });

        mediaRecorder =
            new MediaRecorder(stream);

        audioChunks = [];

        mediaRecorder.ondataavailable =
            function(event) {

                audioChunks.push(event.data);

            };


        mediaRecorder.onstop =
            function() {

                audioBlob =
                    new Blob(audioChunks, {
                        type: "audio/webm"
                    });

                audioURL =
                    URL.createObjectURL(audioBlob);

                document.getElementById(
                    "previewAudio"
                ).src = audioURL;

                document.getElementById(
                    "preview"
                ).style.display = "flex";

                stream.getTracks().forEach(
                    track => track.stop()
                );

            };


        mediaRecorder.start();

        recordingSeconds = 0;

        updateRecordingTime();

        recordingTimer =
            setInterval(updateRecordingTime, 1000);

        document.getElementById(
            "recordingPanel"
        ).classList.add("active");

        document.getElementById(
            "micButton"
        ).classList.add("recording");

        document.getElementById(
            "micButton"
        ).textContent = "⏹";

        document.getElementById(
            "micButton"
        ).onclick = stopRecording;

    }

    catch(error) {

        alert(
            "Microphone permission is required to record audio."
        );

    }

}


/* STOP RECORDING */

function stopRecording() {

    if (!mediaRecorder) {
        return;
    }

    if (mediaRecorder.state !== "inactive") {

        mediaRecorder.stop();

    }

    clearInterval(recordingTimer);

    document.getElementById(
        "recordingPanel"
    ).classList.remove("active");

    document.getElementById(
        "micButton"
    ).classList.remove("recording");

    document.getElementById(
        "micButton"
    ).textContent = "🎙️";

    document.getElementById(
        "micButton"
    ).onclick = startRecording;

}


/* RECORDING TIMER */

function updateRecordingTime() {

    recordingSeconds++;

    let minutes =
        Math.floor(recordingSeconds / 60);

    let seconds =
        recordingSeconds % 60;

    if (seconds < 10) {
        seconds = "0" + seconds;
    }

    document.getElementById(
        "recordTime"
    ).textContent =
        minutes + ":" + seconds;

}


/* CANCEL RECORDING */

function cancelRecording() {

    if (
        mediaRecorder &&
        mediaRecorder.state !== "inactive"
    ) {

        mediaRecorder.stop();

    }

    clearInterval(recordingTimer);

    document.getElementById(
        "recordingPanel"
    ).classList.remove("active");

    document.getElementById(
        "micButton"
    ).classList.remove("recording");

    document.getElementById(
        "micButton"
    ).textContent = "🎙️";

    document.getElementById(
        "micButton"
    ).onclick = startRecording;

    audioChunks = [];

}


/* CANCEL PREVIEW */

function cancelPreview() {

    document.getElementById(
        "preview"
    ).style.display = "none";

    document.getElementById(
        "previewAudio"
    ).pause();

    audioBlob = null;

}


/* SEND AUDIO */

function sendAudio() {

    if (!audioBlob) {
        return;
    }

    const messages =
        document.getElementById("messages");

    const message =
        document.createElement("div");

    message.className =
        "message sent";

    const duration =
        formatTime(recordingSeconds);

    const time =
        new Date().toLocaleTimeString([], {
            hour: "2-digit",
            minute: "2-digit"
        });


    message.innerHTML = `

        <button
            class="audio-play"
            onclick="playRecordedAudio(this)">
            ▶
        </button>

        <div class="audio-info">

            <div class="waveform">

                <span style="height:15px"></span>
                <span style="height:24px"></span>
                <span style="height:18px"></span>
                <span style="height:28px"></span>
                <span style="height:20px"></span>
                <span style="height:13px"></span>
                <span style="height:25px"></span>
                <span style="height:17px"></span>
                <span style="height:27px"></span>
                <span style="height:14px"></span>
                <span style="height:22px"></span>
                <span style="height:11px"></span>

            </div>

            <div class="audio-bottom">

                <span>${duration}</span>

                <span>
                    ${time}
                    <span class="tick">✓✓</span>
                </span>

            </div>

        </div>

    `;


    message.dataset.audio =
        audioURL;

    messages.appendChild(message);

    messages.scrollTop =
        messages.scrollHeight;


    document.getElementById(
        "preview"
    ).style.display = "none";


    /*
        Keep the audio URL available
        for this browser session.
    */

    audioBlob = null;

}


/* PLAY RECORDED AUDIO */

function playRecordedAudio(button) {

    const message =
        button.closest(".message");

    const url =
        message.dataset.audio;

    if (!url) {
        return;
    }

    let audio =
        new Audio(url);

    if (button.textContent === "▶") {

        button.textContent = "⏸";

        audio.play();

        audio.onended = function() {

            button.textContent = "▶";

        };

    }

}


/* PLAY DEMO AUDIO */

function toggleAudio(button) {

    if (button.textContent === "▶") {

        button.textContent = "⏸";

        setTimeout(function() {

            button.textContent = "▶";

        }, 3000);

    }

}


/* FORMAT TIME */

function formatTime(seconds) {

    let minutes =
        Math.floor(seconds / 60);

    let remaining =
        seconds % 60;

    if (remaining < 10) {

        remaining =
            "0" + remaining;

    }

    return minutes + ":" + remaining;

}
